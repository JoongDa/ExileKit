cmake_minimum_required(VERSION 3.25)
project(POEToolbox VERSION 0.3.0 LANGUAGES CXX RC)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "POE Toolbox requires Windows x64 and MSVC.")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /permissive- /utf-8 /EHsc /Zc:__cplusplus)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)

include(CTest)
add_subdirectory(core)
add_library(poetoolbox_platform STATIC platform/windows/logger.cpp platform/windows/paths.cpp platform/windows/utf.cpp platform/windows/launcher.cpp platform/windows/config.cpp platform/windows/dialogs.cpp platform/windows/icon_provider.cpp platform/windows/custom_shortcut.cpp)
target_include_directories(poetoolbox_platform PUBLIC platform/windows)
target_sources(poetoolbox_platform PRIVATE platform/windows/web_metadata.cpp)
target_include_directories(poetoolbox_platform PRIVATE core/src)
target_link_libraries(poetoolbox_platform PUBLIC poetoolbox_core PRIVATE shell32 ole32 windowscodecs user32 gdi32 version uuid winhttp)

add_library(poetoolbox_application STATIC application/application.cpp application/home_actions.cpp application/shortcut_actions.cpp)
target_include_directories(poetoolbox_application PUBLIC application)
target_link_libraries(poetoolbox_application PUBLIC poetoolbox_platform poetoolbox_core)

add_library(poetoolbox_ui STATIC ui/renderer/renderer.cpp ui/shell/main_window.cpp ui/shell/window_actions.cpp ui/shell/shortcut_dialog.cpp ui/pages/library_page.cpp)
target_include_directories(poetoolbox_ui PUBLIC ui)
target_link_libraries(poetoolbox_ui PUBLIC poetoolbox_application PRIVATE d2d1 dwrite dwmapi windowscodecs comctl32)

add_executable(POEToolbox WIN32 apps/toolbox/main.cpp apps/toolbox/app.rc)
target_include_directories(POEToolbox PRIVATE apps/toolbox)
target_link_libraries(POEToolbox PRIVATE poetoolbox_ui)
target_link_options(POEToolbox PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/apps/toolbox/app.manifest)
set_target_properties(POEToolbox PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
install(TARGETS POEToolbox RUNTIME DESTINATION .)
add_custom_target(poetoolbox_resources ALL
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/resources" "$<TARGET_FILE_DIR:POEToolbox>/resources"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/tools/manifests" "$<TARGET_FILE_DIR:POEToolbox>/tools/manifests")
add_dependencies(POEToolbox poetoolbox_resources)
install(DIRECTORY resources DESTINATION .)
install(DIRECTORY tools/manifests DESTINATION tools)
install(FILES tools/manifest.schema.json DESTINATION tools)
install(FILES third_party/nlohmann/LICENSE.MIT DESTINATION licenses/nlohmann)
